cmake_minimum_required(VERSION 3.22)
project(opengothic_turnip_loader LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF)
if(NOT ADRENOTOOLS_SOURCE)
  message(FATAL_ERROR "Set ADRENOTOOLS_SOURCE to the pinned libadrenotools checkout")
endif()
add_subdirectory("${ADRENOTOOLS_SOURCE}" adrenotools)
add_library(ogvulk SHARED wrapper.cpp wrappers.S)
target_link_libraries(ogvulk PRIVATE adrenotools log dl)
target_link_options(ogvulk PRIVATE -Wl,-z,max-page-size=16384)
